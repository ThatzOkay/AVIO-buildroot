kernel=Image

# Force 64-bit mode. Without this, the firmware's arm_64bit auto-detection
# does not reliably kick in for a custom kernel=Image filename, and it was
# silently loading/jumping into this arm64 kernel using the 32-bit boot
# protocol -- the kernel would appear to "load" fine per the firmware log,
# but the ARM cores would crash before executing a single line of kernel
# code, with zero output even from earlycon.
arm_64bit=1

disable_splash=1
boot_delay=0
disable_poe_fan=1

# TTL serial console (see cmdline.txt's console=serial0,115200)
enable_uart=1
dtoverlay=miniuart-bt

# To use an external initramfs file
#initramfs rootfs.cpio.gz

# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1


# Enable autoprobing of Bluetooth driver without need of hciattach/btattach
dtparam=krnbt=on

# Enable dwc2 required for gadget mode
dtoverlay=dwc2,dr_mode=peripheral
